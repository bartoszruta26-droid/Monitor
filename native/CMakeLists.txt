cmake_minimum_required(VERSION 3.20)
project(monitor_core VERSION 0.2.0 LANGUAGES CXX)
option(MONITOR_BUILD_TESTS "Build native tests" ON)
option(MONITOR_BUILD_CLI "Build command-line tools" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(LibArchive REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_library(monitor_core src/regression.cpp src/market_data.cpp src/market_data_adapters.cpp src/market_data_repository.cpp src/market_data_aggregation.cpp src/market_data_chart.cpp src/market_data_regression.cpp src/market_data_calendar.cpp src/xetra_calendar_importer.cpp src/xetra_calendar_source.cpp)
target_include_directories(monitor_core PUBLIC include)
target_link_libraries(monitor_core PUBLIC CURL::libcurl nlohmann_json::nlohmann_json LibArchive::LibArchive LibXml2::LibXml2 ICU::uc ICU::i18n)
target_compile_options(monitor_core PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
if(MONITOR_BUILD_CLI)
 add_executable(monitor-regression src/cli.cpp)
 target_link_libraries(monitor-regression PRIVATE monitor_core)
endif()
if(MONITOR_BUILD_TESTS)
 enable_testing()
 add_executable(monitor-core-tests tests/test_regression.cpp)
 target_link_libraries(monitor-core-tests PRIVATE monitor_core)
 add_test(NAME monitor-core-tests COMMAND monitor-core-tests)
 add_executable(monitor-market-data-tests tests/test_market_data.cpp)
 target_link_libraries(monitor-market-data-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-tests COMMAND monitor-market-data-tests)
 add_executable(monitor-market-data-adapter-tests tests/test_market_data_adapters.cpp)
 target_link_libraries(monitor-market-data-adapter-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-adapter-tests COMMAND monitor-market-data-adapter-tests)
 add_executable(monitor-market-data-repository-tests tests/test_market_data_repository.cpp)
 target_link_libraries(monitor-market-data-repository-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-repository-tests COMMAND monitor-market-data-repository-tests)
 add_executable(monitor-market-data-aggregation-tests tests/test_market_data_aggregation.cpp)
 target_link_libraries(monitor-market-data-aggregation-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-aggregation-tests COMMAND monitor-market-data-aggregation-tests)
 add_executable(monitor-market-data-chart-tests tests/test_market_data_chart.cpp)
 target_link_libraries(monitor-market-data-chart-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-chart-tests COMMAND monitor-market-data-chart-tests)
 add_executable(monitor-market-data-regression-tests tests/test_market_data_regression.cpp)
 target_link_libraries(monitor-market-data-regression-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-regression-tests COMMAND monitor-market-data-regression-tests)
 add_executable(monitor-market-data-calendar-tests tests/test_market_data_calendar.cpp)
 target_compile_definitions(monitor-market-data-calendar-tests PRIVATE MONITOR_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
 target_link_libraries(monitor-market-data-calendar-tests PRIVATE monitor_core)
 add_test(NAME monitor-market-data-calendar-tests COMMAND monitor-market-data-calendar-tests)
 add_executable(monitor-xetra-calendar-source-tests tests/test_xetra_calendar_source.cpp)
 target_compile_definitions(monitor-xetra-calendar-source-tests PRIVATE MONITOR_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
 target_link_libraries(monitor-xetra-calendar-source-tests PRIVATE monitor_core)
 add_test(NAME monitor-xetra-calendar-source-tests COMMAND monitor-xetra-calendar-source-tests)
endif()
