cmake_minimum_required(VERSION 3.20)
project(monitor_core VERSION 0.1.0 LANGUAGES CXX)
option(MONITOR_BUILD_TESTS "Build native tests" ON)
option(MONITOR_BUILD_CLI "Build command-line tools" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(monitor_core src/regression.cpp src/market_data.cpp)
target_include_directories(monitor_core PUBLIC include)
target_compile_options(monitor_core PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
if(MONITOR_BUILD_CLI)
  add_executable(monitor-regression src/cli.cpp)
  target_link_libraries(monitor-regression PRIVATE monitor_core)
endif()
if(MONITOR_BUILD_TESTS)
  enable_testing()
  add_executable(monitor-core-tests tests/test_regression.cpp)
  target_link_libraries(monitor-core-tests PRIVATE monitor_core)
  add_test(NAME monitor-core-tests COMMAND monitor-core-tests)
  add_executable(monitor-market-data-tests tests/test_market_data.cpp)
  target_link_libraries(monitor-market-data-tests PRIVATE monitor_core)
  add_test(NAME monitor-market-data-tests COMMAND monitor-market-data-tests)
endif()
